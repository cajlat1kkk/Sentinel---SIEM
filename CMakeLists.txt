cmake_minimum_required(VERSION 3.24)
project(SentinelSIEM VERSION 0.3.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
set(CMAKE_AUTORCC ON)
include(GNUInstallDirs)
option(SIEM_WITH_DPDK "Build the native DPDK packet sensor" ON)
option(SIEM_BUILD_TESTS "Build integration and unit tests" ON)
find_package(Qt6 6.4 REQUIRED COMPONENTS Core Widgets Network Sql Concurrent)
add_library(siem_core STATIC
  src/core/store.cpp src/core/collector.cpp src/core/scanner.cpp
  src/core/firewall.cpp src/network/packet.cpp src/network/engines.cpp)
target_include_directories(siem_core PUBLIC src)
target_link_libraries(siem_core PUBLIC Qt6::Core Qt6::Network Qt6::Sql Qt6::Concurrent)
add_library(siem_fleet STATIC src/fleet/common.cpp src/fleet/transport.cpp src/fleet/queue.cpp
  src/fleet/repository.cpp src/fleet/service.cpp src/fleet/policy.cpp)
target_link_libraries(siem_fleet PUBLIC siem_core)
add_executable(siem-server src/server_main.cpp)
target_link_libraries(siem-server PRIVATE siem_fleet)
add_executable(siem-fleetctl src/fleetctl_main.cpp)
target_link_libraries(siem-fleetctl PRIVATE siem_fleet)
if(WIN32)
  target_link_libraries(siem_fleet PRIVATE advapi32)
endif()
if(MSVC)
  target_compile_options(siem_fleet PRIVATE /utf-8)
  target_compile_options(siem-server PRIVATE /utf-8)
  target_compile_options(siem-fleetctl PRIVATE /utf-8)
endif()
add_executable(sentinel-siem WIN32 src/main.cpp src/ui/window.cpp src/ui/chart.cpp src/ui/visuals.cpp src/ui/theme.cpp src/ui/ngfw_page.cpp assets/assets.qrc)
target_link_libraries(sentinel-siem PRIVATE siem_core Qt6::Widgets)
add_executable(siem-firewall-helper src/firewall_helper.cpp)
target_link_libraries(siem-firewall-helper PRIVATE siem_core)
if(WIN32)
  target_link_libraries(siem-firewall-helper PRIVATE ole32 oleaut32)
  target_link_libraries(sentinel-siem PRIVATE shell32)
  target_compile_definitions(siem_core PUBLIC NOMINMAX WIN32_LEAN_AND_MEAN)
  if(MSVC)
    add_compile_options(/utf-8)
    target_compile_options(siem_core PRIVATE /utf-8)
    target_compile_options(sentinel-siem PRIVATE /utf-8)
    target_compile_options(siem-firewall-helper PRIVATE /utf-8)
  endif()
endif()
if(SIEM_WITH_DPDK AND CMAKE_SYSTEM_NAME STREQUAL "Linux")
  find_package(PkgConfig REQUIRED)
  pkg_check_modules(DPDK IMPORTED_TARGET libdpdk)
  if(DPDK_FOUND)
    add_executable(siem-dpdk src/network/dpdk_sensor.cpp src/network/packet.cpp src/core/firewall.cpp)
    target_include_directories(siem-dpdk PRIVATE src)
    target_link_libraries(siem-dpdk PRIVATE PkgConfig::DPDK Qt6::Core Qt6::Network)
    install(TARGETS siem-dpdk RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})
  else()
    message(WARNING "DPDK not installed: GUI remains usable; packet sensor will be unavailable")
  endif()
endif()
install(TARGETS sentinel-siem siem-firewall-helper siem-server siem-fleetctl RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})
if(UNIX AND NOT APPLE)
  install(FILES README.md DESTINATION share/doc/sentinel-siem)
  install(DIRECTORY docs samples DESTINATION share/doc/sentinel-siem)
  install(FILES packaging/linux/sentinel-siem.desktop DESTINATION share/applications)
  install(FILES assets/sentinel.svg DESTINATION share/icons/hicolor/scalable/apps)
endif()
if(WIN32)
  qt_generate_deploy_app_script(TARGET sentinel-siem OUTPUT_SCRIPT deploy_script NO_UNSUPPORTED_PLATFORM_ERROR)
  install(SCRIPT ${deploy_script})
  if(SIEM_PG_RUNTIME_DIR)
    # Native Windows paths must not introduce escape sequences in cmake_install.cmake.
    string(REPLACE "\\" "/" siem_pg_runtime_dir "${SIEM_PG_RUNTIME_DIR}")
    install(DIRECTORY "${siem_pg_runtime_dir}/" DESTINATION ${CMAKE_INSTALL_BINDIR}
      FILES_MATCHING PATTERN "*.dll")
  endif()
endif()
if(SIEM_BUILD_TESTS)
  enable_testing()
  find_package(Qt6 REQUIRED COMPONENTS Test)
  add_executable(siem-tests tests/core_tests.cpp)
  target_link_libraries(siem-tests PRIVATE siem_core Qt6::Test)
  if(MSVC)
    target_compile_options(siem-tests PRIVATE /utf-8)
  endif()
  add_executable(siem-ui-tests tests/ui_tests.cpp src/ui/ngfw_page.cpp src/ui/theme.cpp assets/assets.qrc)
  target_link_libraries(siem-ui-tests PRIVATE siem_core Qt6::Widgets Qt6::Test)
  if(WIN32)
    target_link_libraries(siem-ui-tests PRIVATE shell32)
  endif()
  if(MSVC)
    target_compile_options(siem-ui-tests PRIVATE /utf-8)
  endif()
  add_test(NAME ui-profiles COMMAND siem-ui-tests)
  set_tests_properties(ui-profiles PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen" TIMEOUT 30)
  add_test(NAME core COMMAND siem-tests)
  add_test(NAME desktop-smoke COMMAND sentinel-siem --demo --smoke-test)
  add_test(NAME desktop-night-smoke COMMAND sentinel-siem --demo --theme night --smoke-test)
  set_tests_properties(desktop-smoke desktop-night-smoke PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen" TIMEOUT 30)
endif()
set(CPACK_PACKAGE_NAME "sentinel-siem")
set(CPACK_PACKAGE_VENDOR "Sentinel SIEM")
set(CPACK_PACKAGE_CONTACT "Sentinel maintainers")
set(CPACK_PACKAGE_DESCRIPTION_SUMMARY "Native C++ security monitoring, antivirus and firewall console")
set(CPACK_PACKAGE_VERSION ${PROJECT_VERSION})
set(CPACK_PACKAGE_INSTALL_DIRECTORY "Sentinel SIEM")
set(CPACK_DEBIAN_PACKAGE_SECTION "net")
set(CPACK_DEBIAN_PACKAGE_MAINTAINER "Sentinel maintainers")
set(CPACK_DEBIAN_PACKAGE_SHLIBDEPS ON)
set(CPACK_DEBIAN_PACKAGE_DEPENDS "libqt6sql6-psql, libqt6sql6-sqlite, libqt6svg6")
set(CPACK_DEBIAN_PACKAGE_RECOMMENDS "clamav, nftables, policykit-1")
set(CPACK_NSIS_DISPLAY_NAME "Sentinel SIEM")
set(CPACK_NSIS_EXECUTABLES_DIRECTORY "bin")
set(CPACK_PACKAGE_EXECUTABLES "sentinel-siem;Sentinel SIEM")
set(CPACK_NSIS_ENABLE_UNINSTALL_BEFORE_INSTALL ON)
set(CPACK_NSIS_MODIFY_PATH OFF)
include(CPack)
