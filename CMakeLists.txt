cmake_minimum_required(VERSION 3.24)
project(SentinelSIEM VERSION 0.3.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
set(CMAKE_AUTORCC ON)
include(GNUInstallDirs)
option(SIEM_WITH_DPDK "Build the native DPDK packet sensor" ON)
option(SIEM_BUILD_TESTS "Build integration and unit tests" ON)
find_package(Qt6 6.4 REQUIRED COMPONENTS Core Widgets Network Sql Concurrent)
find_package(Qt6 6.4 QUIET COMPONENTS LinguistTools)
add_library(siem_core STATIC
  src/core/store.cpp src/core/collector.cpp src/core/scanner.cpp
  src/core/firewall.cpp src/network/packet.cpp src/network/engines.cpp)
target_include_directories(siem_core PUBLIC src)
target_link_libraries(siem_core PUBLIC Qt6::Core Qt6::Network Qt6::Sql Qt6::Concurrent)
add_library(siem_fleet STATIC src/fleet/common.cpp src/fleet/transport.cpp src/fleet/queue.cpp
  src/fleet/repository.cpp src/fleet/service.cpp src/fleet/policy.cpp src/fleet/agent.cpp
  src/fleet/metrics.cpp src/network/radar.cpp)
target_link_libraries(siem_fleet PUBLIC siem_core)
add_executable(siem-server src/server_main.cpp)
target_link_libraries(siem-server PRIVATE siem_fleet)
add_executable(siem-fleetctl src/fleetctl_main.cpp)
target_link_libraries(siem-fleetctl PRIVATE siem_fleet)
add_executable(siem-agent src/agent_main.cpp)
target_link_libraries(siem-agent PRIVATE siem_fleet)
add_executable(siem-update-verify src/update_verify_main.cpp)
target_link_libraries(siem-update-verify PRIVATE siem_fleet)
if(WIN32)
  target_link_libraries(siem-update-verify PRIVATE crypt32 bcrypt)
else()
  find_package(OpenSSL REQUIRED COMPONENTS Crypto)
  target_link_libraries(siem-update-verify PRIVATE OpenSSL::Crypto)
endif()
if(WIN32)
  target_link_libraries(siem_fleet PRIVATE advapi32 wevtapi)
endif()
if(MSVC)
  target_compile_options(siem_fleet PRIVATE /utf-8)
  target_compile_options(siem-server PRIVATE /utf-8)
  target_compile_options(siem-fleetctl PRIVATE /utf-8)
  target_compile_options(siem-agent PRIVATE /utf-8)
  target_compile_options(siem-update-verify PRIVATE /utf-8)
endif()
add_executable(sentinel-siem WIN32 src/main.cpp src/ui/window.cpp src/ui/chart.cpp src/ui/visuals.cpp
  src/ui/theme.cpp src/ui/ngfw_page.cpp src/ui/fleet_window.cpp src/ui/language.cpp
  src/ui/setup_wizard.cpp src/ui/maintenance_dialog.cpp assets/assets.qrc)
foreach(helper sentinel-maintenance.py fleet-pki.py)
  configure_file(scripts/${helper} maintenance/${helper} COPYONLY)
endforeach()
target_link_libraries(sentinel-siem PRIVATE siem_fleet Qt6::Widgets)
if(Qt6LinguistTools_FOUND)
  set_source_files_properties(translations/sentinel_en_US.ts PROPERTIES
    OUTPUT_LOCATION "${CMAKE_CURRENT_BINARY_DIR}/translations")
  qt_add_translations(sentinel-siem
    TS_FILES translations/sentinel_en_US.ts
    QM_FILES_OUTPUT_VARIABLE sentinel_translation_files)
  install(FILES ${sentinel_translation_files}
    DESTINATION ${CMAKE_INSTALL_DATADIR}/sentinel-siem/translations)
else()
  find_program(SIEM_LRELEASE NAMES lrelease6 lrelease lrelease-qt6 HINTS /usr/lib/qt6/bin)
  if(SIEM_LRELEASE)
    set(sentinel_translation_files "${CMAKE_CURRENT_BINARY_DIR}/translations/sentinel_en_US.qm")
    add_custom_command(OUTPUT ${sentinel_translation_files}
      COMMAND ${CMAKE_COMMAND} -E make_directory "${CMAKE_CURRENT_BINARY_DIR}/translations"
      COMMAND ${SIEM_LRELEASE} "${CMAKE_CURRENT_SOURCE_DIR}/translations/sentinel_en_US.ts"
        -qm ${sentinel_translation_files}
      DEPENDS translations/sentinel_en_US.ts VERBATIM)
    add_custom_target(sentinel-translations ALL DEPENDS ${sentinel_translation_files})
    add_dependencies(sentinel-siem sentinel-translations)
    install(FILES ${sentinel_translation_files}
      DESTINATION ${CMAKE_INSTALL_DATADIR}/sentinel-siem/translations)
  else()
    message(WARNING "Qt LinguistTools/lrelease not found: English catalog will not be built")
  endif()
endif()
# The GUI auto-starts the DPDK sensor on a bundled PCAP sample when no free NIC exists.
add_custom_command(TARGET sentinel-siem POST_BUILD
  COMMAND ${CMAKE_COMMAND} -E make_directory "$<TARGET_FILE_DIR:sentinel-siem>/samples"
  COMMAND ${CMAKE_COMMAND} -E copy_if_different
          "${CMAKE_CURRENT_SOURCE_DIR}/samples/http-lab.pcap"
          "$<TARGET_FILE_DIR:sentinel-siem>/samples/http-lab.pcap")
add_executable(siem-firewall-helper src/firewall_helper.cpp)
target_link_libraries(siem-firewall-helper PRIVATE siem_core)
if(WIN32)
  target_link_libraries(siem-firewall-helper PRIVATE ole32 oleaut32)
  target_link_libraries(sentinel-siem PRIVATE shell32)
  target_compile_definitions(siem_core PUBLIC NOMINMAX WIN32_LEAN_AND_MEAN)
  if(MSVC)
    add_compile_options(/utf-8)
    target_compile_options(siem_core PRIVATE /utf-8)
    target_compile_options(sentinel-siem PRIVATE /utf-8)
    target_compile_options(siem-firewall-helper PRIVATE /utf-8)
  endif()
endif()
if(SIEM_WITH_DPDK AND CMAKE_SYSTEM_NAME STREQUAL "Linux")
  find_package(PkgConfig REQUIRED)
  pkg_check_modules(DPDK IMPORTED_TARGET libdpdk)
  if(DPDK_FOUND)
    add_executable(siem-dpdk src/network/dpdk_sensor.cpp src/network/packet.cpp src/core/firewall.cpp)
    target_include_directories(siem-dpdk PRIVATE src)
    target_link_libraries(siem-dpdk PRIVATE PkgConfig::DPDK Qt6::Core Qt6::Network)
    install(TARGETS siem-dpdk RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})
  else()
    message(WARNING "DPDK not installed: GUI remains usable; packet sensor will be unavailable")
  endif()
endif()
